check assumptions

identify and remove outliers

outliers <- subjectMeans %>% group_by(trialcondition) %>% identify_outliers(subj_mean)
outlier_exclusion <- outliers %>% filter(is.extreme=="TRUE")

subjectMeans2 <- subjectMeans %>% filter(!subject %in% outlier_exclusion)

##2 outliers, no extreme outliers, no exclusions

check normality

##shapiro test
subjectMeans2 %>% group_by(trialcondition) %>% shapiro_test(subj_mean)
##all have p>0.05, but Unsquash-Smashed condition is close at p=0.083. Data satisfies normality condition

##qqplot....plots look fine, not perfect for unsquashables
subjectMeans2_split <- subjectMeans2 %>% separate(trialcondition, into=c("squashability","state"), sep="_")
ggqqplot(subjectMeans2_split, "subj_mean", ggtheme = theme_bw()) + facet_grid(squashability ~ state, labeller = "label_both")

conduct repeated-measures anova

subjectMeans2_split <- subjectMeans2_split %>% ungroup() 

RManova <- anova_test(data = subjectMeans2_split, dv=subj_mean, wid = subject, within = c(squashability, state))
